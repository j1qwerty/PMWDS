```csharp
using AutoMapper;
using MediatR;
using PMWDS.Application.DTOs.Projects;
using PMWDS.Application.Interfaces.Services;
using PMWDS.Domain.Entities;
using PMWDS.Domain.Enums;
namespace PMWDS.Application.Features.Projects.Commands;
public record CreateProjectCommand(
 CreateProjectDto Data) : IRequest<ProjectDto>;
public class CreateProjectCommandHandler
 : IRequestHandler<CreateProjectCommand, ProjectDto>
{
 private readonly IUnitOfWork _uow;
 private readonly ICurrentUserService _currentUser;
 private readonly IAIService _ai;
 private readonly IMapper _mapper;
 public CreateProjectCommandHandler(
 IUnitOfWork uow,
 ICurrentUserService currentUser,
 IAIService ai,
 IMapper mapper)
 {
 _uow = uow;
 _currentUser = currentUser;
 _ai = ai;
 _mapper = mapper;
 }
 public async Task<ProjectDto> Handle(
 CreateProjectCommand request,
 CancellationToken ct)
 {
 var dto = request.Data;
 var project = Project.Create(
 dto.Name, dto.Description,
 dto.Category, dto.Priority,
 dto.DepartmentId, dto.ProjectManagerId,
 dto.PlannedStartDate, dto.PlannedEndDate,
 dto.PlannedBudget, dto.ClientName);
 project.SetCreatedBy(_currentUser.UserId!);
 await _uow.Projects.AddAsync(project, ct);
 await _uow.SaveChangesAsync(ct);
 // Run AI analysis in background
 _ = Task.Run(async () =>
 {
 var health = await _ai
 .AnalyzeProjectHealthAsync(project.Id, ct);
 project.UpdateAIAnalysis(
 health.HealthScore,
 health.DelayRiskScore,
 health.BudgetRiskScore,
 string.Join("; ", health.Insights));
 await _uow.SaveChangesAsync(ct);
 }, ct);
 return _mapper.Map<ProjectDto>(project);
 }
}
```

### Commands/UpdateProjectStatusCommand.cs
```csharp
using MediatR;
using PMWDS.Domain.Enums;
using PMWDS.Application.Interfaces.Services;
using PMWDS.Domain.Entities;
namespace PMWDS.Application.Features.Projects.Commands;
public record UpdateProjectStatusCommand(
 Guid ProjectId,
 ProjectStatus NewStatus,
 string? Justification = null) : IRequest<bool>;
public class UpdateProjectStatusCommandHandler
 : IRequestHandler<UpdateProjectStatusCommand, bool>
{
 private readonly IUnitOfWork _uow;
 private readonly INotificationService _notifications;
 private readonly ICurrentUserService _currentUser;
 public UpdateProjectStatusCommandHandler(
 IUnitOfWork uow,
 INotificationService notifications,
 ICurrentUserService currentUser)
 {
 _uow = uow;
 _notifications = notifications;
 _currentUser = currentUser;
 }
 public async Task<bool> Handle(
 UpdateProjectStatusCommand request,
 CancellationToken ct)
 {
 var project = await _uow.Projects
 .GetByIdAsync(request.ProjectId, ct)
 ?? throw new NotFoundException(
 nameof(Project), request.ProjectId);
 switch (request.NewStatus)
 {
 case ProjectStatus.InProgress:
 project.Start();
 break;
 case ProjectStatus.Completed:
 project.Complete();
 break;
 case ProjectStatus.OnHold:
 project.PutOnHold(
 request.Justification ?? "No reason provided");
 break;
 }
 project.SetModified(_currentUser.UserId!);
 await _uow.SaveChangesAsync(ct);
 return true;
 }
}
```

```csharp
using MediatR;
using AutoMapper;
using PMWDS.Application.DTOs.Projects;
using PMWDS.Application.Interfaces.Services;
using PMWDS.Domain.Enums;
namespace PMWDS.Application.Features.Projects.Queries;
public record GetProjectDashboardQuery(
 Guid? DepartmentId = null) : IRequest<ProjectDashboardDto>;
public record ProjectDashboardDto(
 int TotalProjects,
 int ActiveProjects,
 int CompletedProjects,
 int OverdueProjects,
 int HighRiskProjects,
 double AverageHealthScore,
 decimal TotalBudget,
 decimal TotalActualCost,
 List<ProjectSummaryDto> RecentProjects,
 List<ProjectSummaryDto> AtRiskProjects
);
public record ProjectSummaryDto(
 Guid Id,
 string ProjectCode,
 string Name,
 string Status,
 double ProgressPercentage,
 double AIHealthScore,
 double AIDelayRiskScore,
 int DelayDays,
 DateTime PlannedEndDate
);
public class GetProjectDashboardQueryHandler
 : IRequestHandler<GetProjectDashboardQuery,
 ProjectDashboardDto>
{
 private readonly IUnitOfWork _uow;
 private readonly IMapper _mapper;
 public GetProjectDashboardQueryHandler(
 IUnitOfWork uow, IMapper mapper)
 {
 _uow = uow;
 _mapper = mapper;
 }
 public async Task<ProjectDashboardDto> Handle(
 GetProjectDashboardQuery request,
 CancellationToken ct)
 {
 var projects = request.DepartmentId.HasValue
 ? await _uow.Projects
 .GetByDepartmentAsync(
 request.DepartmentId.Value, ct)
 : await _uow.Projects.GetAllAsync(ct);
 var list = projects.ToList();
 return new ProjectDashboardDto(
 TotalProjects: list.Count,
 ActiveProjects: list.Count(p =>
 p.Status == ProjectStatus.InProgress),
 CompletedProjects: list.Count(p =>
 p.Status == ProjectStatus.Completed),
 OverdueProjects: list.Count(p =>
 p.GetDelayDays() > 0),
 HighRiskProjects: list.Count(p =>
 p.AIDelayRiskScore >= 0.7),
 AverageHealthScore: list.Any()
 ? list.Average(p => p.AIHealthScore)
 : 0,
 TotalBudget: list.Sum(p => p.PlannedBudget),
 TotalActualCost: list.Sum(p => p.ActualCost),
 RecentProjects: list.OrderByDescending(p => p.CreatedDate)
 .Take(5)
 .Select(p => _mapper.Map<ProjectSummaryDto>(p))
 .ToList(),
 AtRiskProjects: list
 .Where(p => p.AIDelayRiskScore >= 0.7)
 .OrderByDescending(p => p.AIDelayRiskScore)
 .Take(10)
 .Select(p => _mapper.Map<ProjectSummaryDto>(p))
 .ToList()
 );
 }
}
```

```csharp
using MediatR;
using PMWDS.Application.DTOs.Tasks;
using AutoMapper;
using PMWDS.Application.Interfaces.Services;
using PMWDS.Domain.Entities;
namespace PMWDS.Application.Features.Tasks.Commands;
public record CreateTaskCommand(
 CreateTaskDto Data) : IRequest<TaskDto>;
public class CreateTaskCommandHandler
 : IRequestHandler<CreateTaskCommand, TaskDto>
{
 private readonly IUnitOfWork _uow;
 private readonly IAIService _ai;
 private readonly INotificationService _notifications;
 private readonly ICurrentUserService _currentUser;
 private readonly IMapper _mapper;
 public CreateTaskCommandHandler(
 IUnitOfWork uow, IAIService ai,
 INotificationService notifications,
 ICurrentUserService currentUser,
 IMapper mapper)
 {
 _uow = uow;
 _ai = ai;
 _notifications = notifications;
 _currentUser = currentUser;
 _mapper = mapper;
 }
 public async Task<TaskDto> Handle(
 CreateTaskCommand request,
 CancellationToken ct)
 {
 var dto = request.Data;
 var task = ProjectTask.Create(
 dto.ProjectId, dto.Title,
 dto.Description, dto.Priority,
 dto.StartDate, dto.DueDate,
 dto.EstimatedHours,
 dto.MilestoneId, dto.ParentTaskId);
 task.SetCreatedBy(_currentUser.UserId!);
 // Add dependencies
 if (dto.DependencyIds?.Any() == true)
 foreach (var depId in dto.DependencyIds)
 task.AddDependency(TaskDependency.Create(
 depId, task.Id));
 await _uow.Tasks.AddAsync(task, ct);
 // AI: get optimal assignee suggestion
 var aiRec = await _ai
 .GetOptimalAssigneeAsync(task.Id, ct);
 // Auto-assign or notify manager
 if (!string.IsNullOrEmpty(dto.AssignedToUserId))
 {
 task.AssignTo(
 dto.AssignedToUserId,
 _currentUser.UserId!);
 await _notifications.SendTaskAssignmentAlertAsync(
 task.Id, dto.AssignedToUserId, ct);
 }
 // Attach AI prediction
 var delay = await _ai.PredictTaskDelayAsync(task.Id, ct);
 task.UpdateAIPrediction(
 delay.DelayProbability,
 delay.PredictedCompletionDate,
 string.Join(", ", delay.ContributingFactors),
 aiRec.RecommendedUserId);
 await _uow.SaveChangesAsync(ct);
 return _mapper.Map<TaskDto>(task);
 }
}
```

```csharp
using MediatR;
using PMWDS.Application.Interfaces.Services;
using PMWDS.Domain.Entities;
namespace PMWDS.Application.Features.Tasks.Commands;
public record AssignTaskCommand(
 Guid TaskId,
 string AssigneeId,
 bool UseAIRecommendation = false)
 : IRequest<bool>;
public class AssignTaskCommandHandler
 : IRequestHandler<AssignTaskCommand, bool>
{
 private readonly IUnitOfWork _uow;
 private readonly IAIService _ai;
 private readonly INotificationService _notifications;
 private readonly ICurrentUserService _currentUser;
 public AssignTaskCommandHandler(
 IUnitOfWork uow, IAIService ai,
 INotificationService notifications,
 ICurrentUserService currentUser)
 {
 _uow = uow;
 _ai = ai;
_notifications = notifications;
 _currentUser = currentUser;
 }
 public async Task<bool> Handle(
 AssignTaskCommand request,
 CancellationToken ct)
 {
 var task = await _uow.Tasks
 .GetByIdAsync(request.TaskId, ct)
 ?? throw new NotFoundException(
 nameof(ProjectTask), request.TaskId);
 var assigneeId = request.AssigneeId;
 if (request.UseAIRecommendation)
 {
 var rec = await _ai
 .GetOptimalAssigneeAsync(request.TaskId, ct);
 assigneeId = rec.RecommendedUserId;
 }
 task.AssignTo(assigneeId, _currentUser.UserId!);
 task.SetModified(_currentUser.UserId!);
 await _uow.SaveChangesAsync(ct);
 await _notifications.SendTaskAssignmentAlertAsync(
 task.Id, assigneeId, ct);
 return true;
 }
}
```
