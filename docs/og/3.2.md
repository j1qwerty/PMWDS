``` csharp
using PMWDS.Application.Interfaces.Services;
namespace PMWDS.Infrastructure.Jobs;
public interface IDeadlineCheckerJob
{
 Task ExecuteAsync(CancellationToken ct);
}
public class DeadlineCheckerJob : IDeadlineCheckerJob
{
 private readonly IUnitOfWork _uow;
 private readonly INotificationService _notifications;
 public DeadlineCheckerJob(
 IUnitOfWork uow,
 INotificationService notifications)
 {
 _uow = uow;
 _notifications = notifications;
 }
 public async Task ExecuteAsync(CancellationToken ct)
 {
 var overdueTasks = await _uow.Tasks
 .GetOverdueTasksAsync(ct);
 foreach (var task in overdueTasks)
 {
 var daysRemaining =
 (int)(task.DueDate - DateTime.UtcNow)
 .TotalDays;
 // Send reminders for 7, 3, 1 days
 if (daysRemaining is 7 or 3 or 1)
 await _notifications
 .SendDeadlineReminderAsync(
 task.Id, daysRemaining, ct);
 // Auto-escalate if already overdue
 if (daysRemaining < 0 && !task.IsEscalated)
 {
 task.Escalate();
 await _uow.SaveChangesAsync(ct);
 await _notifications
 .SendEscalationAlertAsync(
 task.Id,
 task.EscalationLevel, ct);
 }
 }
 }
}
```

``` csharp
using PMWDS.Application.Interfaces.Services;
namespace PMWDS.Infrastructure.Jobs;
public interface IEscalationCheckerJob
{
 Task ExecuteAsync(CancellationToken ct);
}
public class EscalationCheckerJob : IEscalationCheckerJob
{
 private readonly IUnitOfWork _uow;
 private readonly INotificationService _notifications;
 public EscalationCheckerJob(
 IUnitOfWork uow,
 INotificationService notifications)
 {
 _uow = uow;
 _notifications = notifications;
 }
 public async Task ExecuteAsync(CancellationToken ct)
 {
 var escalatedTasks = await _uow.Tasks
 .GetEscalatedTasksAsync(ct);
 foreach (var task in escalatedTasks)
 {
 // Re-escalate if still not resolved after 48 hrs
 if (task.EscalatedDate.HasValue
 && (DateTime.UtcNow - task.EscalatedDate.Value)
 .TotalHours >= 48
 && task.Status != TaskStatus.Completed)
 {
 task.Escalate();
 await _uow.SaveChangesAsync(ct);
 await _notifications
 .SendEscalationAlertAsync(
 task.Id,
 task.EscalationLevel, ct);
 }
 }
 }
}
```

``` csharp
 IAIService ai,
 ILogger<AIModelTrainingJob> logger)

 _logger = logger;
 }

 public async Task ExecuteAsync(CancellationToken ct)
 {
 _logger.LogInformation(
 "[AIModelTrainingJob] Starting model training: {Time}",
 DateTime.UtcNow);
 try
 {
 await _ai.TrainModelsAsync(ct);
 _logger.LogInformation(
 "[AIModelTrainingJob] Training complete: {Time}",
 DateTime.UtcNow);
 }
 catch (Exception ex)
 {
 _logger.LogError(ex,
 "[AIModelTrainingJob] Training failed.");
 }
 }
```

``` csharp
using PMWDS.Application.Interfaces.Services;
namespace PMWDS.Infrastructure.Jobs;
public interface IScheduledReportJob
{
 Task ExecuteAsync(CancellationToken ct);
}
public class ScheduledReportJob : IScheduledReportJob
{
 private readonly IReportService _reports;
 private readonly IEmailService _email;
 private readonly IUnitOfWork _uow;
 public ScheduledReportJob(
 IReportService reports,
 IEmailService email,
 IUnitOfWork uow)
 {
 _reports = reports;
 _email = email;
 _uow = uow;
 }
 public async Task ExecuteAsync(CancellationToken ct)
 {
 var projects = await _uow.Projects
 .GetByStatusAsync(
 Domain.Enums.ProjectStatus.InProgress, ct);
 foreach (var project in projects)
 {
 var pdfBytes = await _reports
 .GenerateProjectStatusReportAsync(
 project.Id, "pdf", ct);
 // Email to project manager
 var manager = await _uow.Users
 .GetByIdAsync(
 Guid.Parse(project.ProjectManagerId), ct);
 if (manager == null) continue;
 // In production: attach pdf bytes to email
 await _email.SendEmailAsync(
 manager.Email,
 $"Weekly Status Report — {project.Name}",
 $"<p>Please find the weekly status report " +
 $"for <b>{project.Name}</b> attached.</p>",
 ct);
 }
 }
}
```
