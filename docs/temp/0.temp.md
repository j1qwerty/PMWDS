🚀PMWDS 2.0 — Complete Solution Setup Guide
🚀part 1 — SOLUTION & PROJECT STRUCTURE
Step 1 — Create Solution Open Terminal / Command Prompt and run:

# Create root folder
mkdir PMWDS
cd PMWDS
# Create solution
dotnet new sln -n PMWDS
# Create all projects
dotnet new webapi -n PMWDS.API --no-openapi
dotnet new classlib -n PMWDS.Domain
dotnet new classlib -n PMWDS.Application
dotnet new classlib -n PMWDS.Persistence
dotnet new classlib -n PMWDS.Infrastructure
dotnet new classlib -n PMWDS.AI
# Add all projects to solution
dotnet sln add PMWDS.API/PMWDS.API.csproj
dotnet sln add PMWDS.Domain/PMWDS.Domain.csproj
dotnet sln add PMWDS.Application/PMWDS.Application.csproj
dotnet sln add PMWDS.Persistence/PMWDS.Persistence.csproj
dotnet sln add PMWDS.Infrastructure/PMWDS.Infrastructure.csproj
dotnet sln add PMWDS.AI/PMWDS.AI.csproj

```
dotnet new sln -n PMWDS
mkdir -p src/Core
mkdir -p src/Infrastructure
mkdir -p src/AI
mkdir -p src/Presentation
mkdir -p tests

dotnet new classlib -n PMWDS.Domain -o src/Core/PMWDS.Domain
dotnet new classlib -n PMWDS.Application -o src/Core/PMWDS.Application

dotnet new classlib -n PMWDS.Infrastructure -o src/Infrastructure/PMWDS.Infrastructure
dotnet new classlib -n PMWDS.Persistence -o src/Infrastructure/PMWDS.Persistence

dotnet new webapi -n PMWDS.API -o src/Presentation/PMWDS.API
dotnet new mvc -n PMWDS.Web -o src/Presentation/PMWDS.Web
dotnet new xunit -n PMWDS.Domain.Tests -o tests/PMWDS.Domain.Tests
dotnet new xunit -n PMWDS.Application.Tests -o tests/PMWDS.Application.Tests
dotnet new xunit -n PMWDS.Integration.Tests -o tests/PMWDS.Integration.Tests

dotnet new xunit -n PMWDS.Domain.Tests -o tests/PMWDS.Domain.Tests
dotnet new xunit -n PMWDS.Application.Tests -o tests/PMWDS.Application.Tests
dotnet new xunit -n PMWDS.Integration.Tests -o tests/PMWDS.Integration.Tests

Get-ChildItem -Recurse -Filter *.csproj | ForEach-Object { dotnet sln add $_.FullName }

dotnet add src/Core/PMWDS.Application reference src/Core/PMWDS.Domain

dotnet add src/Infrastructure/PMWDS.Infrastructure reference src/Core/PMWDS.Application

dotnet add src/Presentation/PMWDS.API reference src/Core/PMWDS.Application
dotnet add src/Presentation/PMWDS.API reference src/Infrastructure/PMWDS.Persistence
```
